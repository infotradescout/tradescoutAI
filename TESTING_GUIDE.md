## Automated Test Script

### Create: `test-api.ps1`

```powershell
# Test Suite Runner
$baseUrl = "http://localhost:5000/api/assistant"

function Test-Query {
    param(
        [string]$message,
        [string]$testName,
        [string]$expectedLayer = "any"
    )
    
    Write-Host "`n📋 Test: $testName" -ForegroundColor Cyan
    Write-Host "Query: $message"
    
    $body = @{message = $message} | ConvertTo-Json
    $response = Invoke-RestMethod -Uri $baseUrl -Method Post -Body $body -ContentType "application/json"
    
    Write-Host "Response:" -ForegroundColor Yellow
    $response | ConvertTo-Json | Write-Host
    
    if ($response.knowledge.layer) {
        Write-Host "✓ Layer $($response.knowledge.layer) - $($response.knowledge.confidence) confidence" -ForegroundColor Green
    }
}

# Run tests
Test-Query "hurricane preparation" "Layer 1 - Manual Override" "1"
Test-Query "Find contractors" "Layer 2 - Cache Hit" "2"
Test-Query "Unknown topic XYZ" "Layer 4 - Internet Search" "4"

# Health check
Write-Host "`n📋 Health Check" -ForegroundColor Cyan
$health = Invoke-RestMethod -Uri "$baseUrl/health"
$health | ConvertTo-Json | Write-Host
```

**Run:**
```powershell
.\test-api.ps1
```
